### [P1005 [NOIP 2007 提高组\] 矩阵取数游戏 - 洛谷](https://www.luogu.com.cn/problem/P1005)

#### 思路

先看范围：$1\le m\le 80$，所以$2^{80}$只能用高精度存储计算。

由于在不同行取数互不影响，所以我们可以单独求解每一行取数的最优解，最后再相加。

每一次取数，不是取左边的，就是取右边的，所以我想到了递归实现dfs。可惜，dfs超时了。

然后注意到，如果用left和right分别表示剩余数字的区间左、右端点，那么left单增，right单减，所以我想到了可以列出dp的递推公式，进而使用dp算法。

#### 题解

首先需要用到非负数的高精度乘法和加法，我用的是```vector<short>```模拟大数运算。

其次，动态规划：对于每一行，我们定义状态 `dp[left][right]`表示在当前行中，剩余区间为 [left, right] 时能够获得的最大得分。

剩余区间最优解=max(取左边的最优解,取右边的最优解)

#if 1
#include<bits/stdc++.h>
#define ll long long
#define sh short
using namespace std;
vector<sh> mul(vector<sh> a, vector<sh> b)
{
    vector<sh> ans(a.size() + b.size() + 1);
    for (int i = 0; i < a.size(); i++)
    {
        for (int j = 0; j < b.size(); j++)
        {
            sh temp = a[i] * b[j];
            ans[i + j] += temp % 10;
            ans[i + j + 1] += temp / 10 + ans[i + j] / 10;
            ans[i + j] %= 10;
            ans[i + j + 2] += ans[i + j + 1] / 10;
            ans[i + j + 1] %= 10;
        }
    }
    while (ans[ans.size() - 1] == 0 && ans.size() > 1)
    {
        ans.pop_back();
    }
    return ans;
}
vector<sh> add(vector<sh> a, vector<sh> b)
{
    ll al = a.size();
    ll bl = b.size();
    ll ans_size = max(al, bl);
    if (al > bl)
    {
        for (int i = 0; i < al - bl; i++)
        {
            b.push_back(0);
        }
    }
    else if (bl > al)
    {
        for (int i = 0; i < bl - al; i++)
        {
            a.push_back(0);
        }
    }

    vector<sh> ans(ans_size + 1, 0);
    for (int i = 0; i < ans_size; i++)
    {
        ans[i] += a[i] + b[i];
        if (ans[i] >= 10)
        {
            ans[i + 1] += ans[i] / 10;
            ans[i] %= 10;
        }
    }
    if (ans[ans_size] == 0)
    {
        ans.pop_back();
    }
    return ans;


}

//当a>b时返回真，否则为假
bool compare(vector<sh>a, vector<sh>b)
{
    if (a.size() != b.size()) { return a.size() > b.size(); }
    else
    {
        for (int i = a.size() - 1; i >= 0; i--)
        {
            if (a[i] != b[i]) { return a[i] > b[i]; }
        }
        return 0;
    }
}
vector<sh>tran(int a)
{
    if (a == 0) { return { 0 }; }
    vector<sh>result;
    while (a)
    {
        result.push_back(a % 10);
        a /= 10;
    }
    return result;
}

    for (int k = 0; k < n; k++)
    {
        auto temp = board[k];
        vector<vector<vector<sh>>> dp(m + 2, vector<vector<sh>>(m + 2, {0}));
        vector<sh>max_sum = { 0 };
        for (int right = m; right >= 0; right--)

            for (int left = 1; left <= m + 1 && left < right + 2; left++)

                        add(dp[left - 1][right], mul(tran(temp[left - 2]), two[m - right + left - 1])));
                }
                if (left == right + 1) { if (compare(dp[left][right], max_sum)) { max_sum = dp[left][right]; } }
            }
        }
        ans = add(ans, max_sum);
    }
