[P1006 [NOIP 2008 提高组] 传纸条 - 洛谷](https://www.luogu.com.cn/problem/P1006)

#### 思路

用动态规划

#### 题解

可以看作从同一个人同时传了两次，只是路线不能有重合。既然两张纸条同时出发且每一步都只能向右或向下，那么他们在第$step$步时，走过的总步数是一样的。设第一个人的位置是 $(x_1, y_1)$，第二个人的位置是 $(x_2, y_2)$，则有$ x_1 + y_1 = x_2 + y_2 = step + 2$

那么我就令$dp[step][x_1][x_2]$表示走了走了$step$步，横坐标分别是$x_1,x_2$

#### 代码

```c++
#if 1
#include<bits/stdc++.h>
#define vec vector
using namespace std;
int m, n;
int board[51][51];//board数组都采用1-base索引，board[y][x]
int dp[99][51][51];//表示dp[step][x1][x2]
signed main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr); cout.tie(nullptr);
    cin >> m >> n;
    for (int i = 1; i <= m; i++)
    {
        for (int j = 1; j <= n; j++)
        {
            cin >> board[i][j];
        }
    }

    for (int step = 0; step <= m + n - 2; step++)
    {
        for (int i = 1; i <= 50; i++)
        {
            for (int j = 1; j <= 50; j++)
            {
                dp[step][i][j] = -2147483648;
            }
        }
    }

    dp[0][1][1] = 0;
    for (int step = 1; step <= m + n - 2; step++)
    {
        for (int x1 = max(1, step - m + 2); x1 <= min(n,step+1); x1++)
        {
            for (int x2 = max(1, step - m + 2); x2 <= min(n, step + 1); x2++)
            {
                if (x1 == x2 && step != m + n - 2) { continue; }
                int t1 = dp[step - 1][x1 - 1][x2 - 1];
                int t2 = dp[step - 1][x1][x2 - 1];
                int t3 = dp[step - 1][x1 - 1][x2];
                int t4 = dp[step - 1][x1][x2];
                int max_num = max({ t1,t2,t3,t4 });
                if (x1 == x2) { dp[step][x1][x2] = max_num + board[step - x1 + 2][x1]; }
                else { dp[step][x1][x2] = max_num + board[step - x1 + 2][x1] + board[step - x2 + 2][x2]; }
            }
        }
    }
    cout << dp[m + n - 2][n][n] << endl;
    return 0;
}
#endif
```
