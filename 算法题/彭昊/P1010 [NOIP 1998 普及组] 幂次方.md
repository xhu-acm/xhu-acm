```c++
#if 1
#include<bits/stdc++.h>
#define int long long
using namespace std;
string fun(int a)
{
    if (a == 0)
    {
        return"";
    }
    if (a == 1)
    {
        return "2(0)";
    }
    if (a == 2)
    {
        return "2";
    }
    string s;
    for (int i = 0; (a >> i)!= 0; i++)
    {
        if (((a >> i) & 1) != 0)
        {
            string temp;
            
            if (i==0)
            {
                temp = "2(0)+";
            }
            else if (i == 1)
            {
                temp = "2+";
            }
            else
            {
                
                temp = ("2(" + fun(i) + ")+");
                
            }
            temp += s;
            s = temp;
        }
    }
    while (s[s.size()-1]=='+')
    {
        s.pop_back();
    }
    return s;
}
signed main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr); cout.tie(nullptr);
    int a;
    cin >> a;
    cout << fun(a);
    

    system("pause");
    return 0;
}
#endif
```
