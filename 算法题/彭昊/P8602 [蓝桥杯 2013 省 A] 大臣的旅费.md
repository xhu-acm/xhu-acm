### [P8602 [蓝桥杯 2013 省 A\] 大臣的旅费 - 洛谷](https://www.luogu.com.cn/problem/P8602)

#### 思路

先从某一个点出发，利用dfs找出最远的城市，再从刚刚找到的最远的城市出发，利用dfs找出最远距离就可以了

#### 代码

```c++
#if 1
#include<bits/stdc++.h>
#define int long long
using namespace std;
vector<map<int, int>> city;
int len = 0;
int max_len = 0;
int farest = 0;
int deep = 0;
void dfs(int index,int last)
{
    if (city[index].size() == 1 && deep != 0)
    {
        if (max_len < len)
        {
            max_len = len;
            farest = index;
            return;
        }
    }
    for (auto i : city[index])
    {
        if (i.first == last) { continue; }
        len += i.second;
        deep++;
        dfs(i.first,index);
        len -= i.second;
        deep--;
    }
}
signed main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr); cout.tie(nullptr);
    int n; cin >> n;
    city.resize(n + 1);
    for (int i = 1; i < n; i++)
    {
        int p, q, d; cin >> p >> q >> d;
        city[p][q] = d;
        city[q][p] = d;
    }
    dfs(1,-1);
    len = 0;
    max_len = 0;
    deep = 0;
    int start = farest;
    farest = 0;
    dfs(start,-1);
    //cout << max_len << endl;
    cout << ((max_len + 1) * max_len) / 2 + max_len * 10 << endl;
    return 0;
}
#endif
```
