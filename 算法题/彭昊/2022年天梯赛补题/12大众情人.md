## 大众情人

人与人之间总有一点距离感。我们假定两个人之间的亲密程度跟他们之间的距离感成反比，并且距离感是单向的。例如小蓝对小红患了单相思，从小蓝的眼中看去，他和小红之间的距离为 1，只差一层窗户纸；但在小红的眼里，她和小蓝之间的距离为 108000，差了十万八千里…… 另外，我们进一步假定，距离感在认识的人之间是可传递的。例如小绿觉得自己跟小蓝之间的距离为 2，则即使小绿并不直接认识小红，我们也默认小绿早晚会认识小红，并且因为跟小蓝很亲近的关系，小绿会觉得自己跟小红之间的距离为 1+2=3。当然这带来一个问题，如果小绿本来也认识小红，或者他通过其他人也能认识小红，但通过不同渠道推导出来的距离感不一样，该怎么算呢？我们在这里做个简单定义，就将小绿对小红的距离感定义为所有推导出来的距离感的最小值。

一个人的异性缘不是由最喜欢他/她的那个异性决定的，而是由对他/她最无感的那个异性决定的。我们记一个人 $i$ 在一个异性 $j$ 眼中的距离感为$ D_{ij}$；将$i$ 的“异性缘”定义为 $1/max_{j*∈S(i)}{D_{ij}}$，其中 $S(i)$ 是相对于 $i$ 的所有异性的集合。那么“大众情人”就是异性缘最好（值最大）的那个人。

本题就请你从给定的一批人与人之间的距离感中分别找出两个性别中的“大众情人”。

### 输入格式：

输入在第一行中给出一个正整数 $N(≤500)$，为总人数。于是我们默认所有人从$1$到 $N$ 编号。

随后 *N* 行，第 $i$ 行描述了编号为 $i$ 的人与其他人的关系，格式为：

```
性别 K 朋友1:距离1 朋友2:距离2 …… 朋友K:距离K
```

其中 `性别` 是这个人的性别，`F` 表示女性，`M` 表示男性；`K`（<*N* 的非负整数）为这个人直接认识的朋友数；随后给出的是这 `K` 个朋友的编号、以及这个人对该朋友的距离感。距离感是不超过 106 的正整数。

题目保证给出的关系中一定两种性别的人都有，不会出现重复给出的关系，并且每个人的朋友中都不包含自己。

### 输出格式：

第一行给出自身为女性的“大众情人”的编号，第二行给出自身为男性的“大众情人”的编号。如果存在并列，则按编号递增的顺序输出所有。数字间以一个空格分隔，行首尾不得有多余空格。

### 输入样例：

```
6
F 1 4:1
F 2 1:3 4:10
F 2 4:2 2:2
M 2 5:1 3:2
M 2 2:2 6:2
M 2 3:1 2:5

```

### 输出样例：

```
2 3
4

```

## 思路

我最开始想到了对每一个点单独使用迪杰斯特拉算法，结果最后一个测试点超时。然后问AI，才知道弗洛伊德算法可以在$O(n^3)$的时间复杂度下求出全源最短路径

## 代码

#### 弗洛伊德版本

（成功AC）

```c++
#if 1
#define _CRT_SECURE_NO_WARNINGS
#include<bits/stdc++.h>
using namespace std;
const int inf = 1000000000;
signed main()
{
    int n; cin >> n;
    vector<vector<int>>person(n + 1, vector<int>(n + 1, inf));
    for (int i = 1; i <= n; i++)
    {
        person[i][i] = 0;
    }
    vector<char>gender(n + 1, 'x');
    for (int i = 1; i <= n; i++)
    {
        char g; cin >> g;
        gender[i] = g;
        int times;
        cin >> times;
        for (int j = 0; j < times; j++)
        {
            int a = 0, b = 0;
            scanf("%d:%d", &a, &b);
            person[i][a] = b;
        }
    }


    //弗洛伊德算法
    for (int i = 1; i <= n; i++)
    {
        for (int j = 1; j <= n; j++)
        {
            if (j == i) { continue; }
            for (int k = 1; k <= n; k++)
            {
                if (k == i || k == j) { continue; }
                if (person[j][i] + person[i][k] < person[j][k])
                {
                    person[j][k] = person[j][i] + person[i][k];
                }
            }
        }
    }

    int F_min=inf, M_min=inf;
    vector<int> F_lover, M_lover;
    for (int i = 1; i <= n; i++)
    {
        char temp_gender = gender[i];
        int len_max = -1;
        if (temp_gender == 'F')
        {
            for (int j = 1; j <= n; j++)
            {
                if (gender[j] != temp_gender && len_max < person[j][i])
                {
                    len_max = person[j][i];
                }
            }
            if (F_min > len_max)
            {
                F_lover.clear();
                F_lover.push_back(i);
                F_min = len_max;
            }
            else if(F_min==len_max)
            {
                F_lover.push_back(i);
            }
        }
        else
        {
            for (int j = 1; j <= n; j++)
            {
                if (gender[j] != temp_gender && len_max < person[j][i])
                {
                    len_max = person[j][i];
                }
            }
            if (M_min > len_max)
            {
                M_lover.clear();
                M_lover.push_back(i);
                M_min = len_max;
            }
            else if (M_min == len_max)
            {
                M_lover.push_back(i);
            }
        }
    }
    for (int i = 0; i < F_lover.size(); i++)
    {
        cout << F_lover[i];
        if (i != F_lover.size() - 1)
        {
            cout << " ";
        }
        else
        {
            cout << endl;
        }

    }
    for (int i = 0; i < M_lover.size(); i++)
    {
        cout << M_lover[i];
        if (i != M_lover.size() - 1)
        {
            cout << " ";
        }
        else
        {
            cout << endl;
        }

    }
    return 0;
}
#endif
```

#### 迪杰斯特拉算法

（最后一个测试点超时）

```c++
#if 1
#define _CRT_SECURE_NO_WARNINGS
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> pp;//<距离，人>
const int inf = 2147483647;
struct person
{
    char gender = 'x';
    vector<pp> p;
};
signed main()
{
    int n; cin >> n;
    vector<person> all(n + 1);
    vector<ll>far_max(n + 1, -1);
    for (int i = 1; i <= n; i++)
    {
        char gender; cin >> gender;
        all[i].gender = gender;
        int times; cin >> times;
        //all[i].p.resize(times);
        for (int j = 0; j < times; j++)
        {
            int a, b;
            scanf("%d:%d", &a, &b);
            all[a].p.push_back({ b,i });
        }
    }
    for (int i = 1; i <= n; i++)
    {
        priority_queue<pp,vector<pp>,greater<pp>> pq;
        char gender = all[i].gender;
        pq.push({ 0,i });
        vector<int> dis(n + 1, inf);
        while (!pq.empty())
        {
            int len = pq.top().first;
            int index = pq.top().second;
            pq.pop();
            if (len < dis[index])
            {
                dis[index] = len;
                for (int j = 0; j < all[index].p.size(); j++)
                {
                    auto temp = all[index].p[j];
                    temp.first += len;
                    if (temp.first < dis[temp.second])
                    {
                        pq.push({ temp.first,temp.second });
                    }
                }
            }
        }
        int max_len = -1;
        for (int w = 1; w <= n; w++)
        {
            if (all[w].gender != gender && max_len <= dis[w])
            {
                max_len = dis[w];
            }
        }
        far_max[i] = max_len;
    }
    int man = inf, woman = inf;
    vector<int>man_index, woman_index;
    for (int i = 1; i <= n; i++)
    {
        if (all[i].gender == 'F')
        {
            if (far_max[i] < woman)
            {
                woman = far_max[i];
                woman_index.clear();
                woman_index.push_back(i);
            }
            else if (far_max[i] == woman)
            {
                woman_index.push_back(i);
            }
        }
        else
        {
            if (far_max[i] < man)
            {
                man = far_max[i];
                man_index.clear();
                man_index.push_back(i);
            }
            else if (far_max[i] == man)
            {
                man_index.push_back(i);
            }
        }
    }

    for (int i = 0; i < woman_index.size(); i++)
    {
        cout << woman_index[i];
        if (i != woman_index.size() - 1)
        {
            cout << " ";
        }
        else
        {
            cout << endl;
        }
    }
    for (int i = 0; i < man_index.size(); i++)
    {
        cout << man_index[i];
        if (i != man_index.size() - 1)
        {
            cout << " ";
        }
        else
        {
            cout << endl;
        }
    }
    return 0;
}

#endif
```
