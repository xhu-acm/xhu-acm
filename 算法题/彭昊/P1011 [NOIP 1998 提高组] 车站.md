[P1011 [NOIP 1998 提高组] 车站 - 洛谷](https://www.luogu.com.cn/problem/P1011)

#### 思路

这种题要先找规律

#### 题解

假设第2站上下车人数为$b$，那么从第3站开始，a与b前的系数都可以看作是斐波那契数列的变形。

#### 代码

```c++
#if 1
#include<bits/stdc++.h>
#define int long long
using namespace std;
signed main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr); cout.tie(nullptr);
    int a, n, m, x, b;
    cin >> a >> n >> m >> x;
    int fb[22];
    fb[1] = 1; fb[2] = 0;
    for (int i = 3; i <= 21; i++) { fb[i] = fb[i - 1] + fb[i - 2]; }
    int a_num = fb[n - 1]+1, b_num = fb[n]-1;
    b = (m - a_num * a) / b_num;
    if (x == 1 || x == 2)
    {
        cout << a;
    }
    else
    {
        cout << (fb[x] + 1) * a + (fb[x + 1] - 1) * b;
    }


    system("pause");
    return 0;
}
#endif
```
