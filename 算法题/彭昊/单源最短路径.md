朴素版

```c++
#if 1
#include<bits/stdc++.h>
#define int long long

using namespace std;
const int inf = 2147483647;
signed main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr); cout.tie(nullptr);
    int n, m, s;
    cin >> n >> m >> s;

    vector<map<int, int>>point(n + 1);
    for (int i = 1; i <= m; i++)
    {
        int u, v, w;
        cin >> u >> v >> w;
        if (point[u].count(v) == 1)
        {
            if (point[u][v] > w)
            {
                point[u][v] = w;
            }
        }
        else
        {
            point[u][v] = w;
        }
    }
    vector<int> has(n + 1, inf);
    vector<int>road(n + 1, inf);
    road[s] = 0;
    for (int i = 1; i < n; i++)
    {
        int min_ = inf;
        int index = 0;
        for (int j = 1; j <= n; j++)
        {
            if (has[j] != inf) { continue; }
            if (min_ > road[j])
            {
                min_ = road[j];
                index = j;
            }
        }
        has[index] = min_;
        for (auto next : point[index])
        {
            if (has[next.first] != inf) { continue; }
            if (min_ + next.second < road[next.first])
            {
                road[next.first] = min_ + next.second;
            }
        }
        if (i == n - 1)
        {
            min_ = inf;
            index = 0;
            for (int j = 1; j <= n; j++)
            {
                if (has[j] != inf) { continue; }
                if (min_ > road[j])
                {
                    min_ = road[j];
                    index = j;
                }
            }
            has[index] = min_;
        }
    }
    for (int i = 1; i <= n; i++)
    {
        cout << road[i] << " ";
    }
    return 0;
}
#endif
```

优先队列优化版

```c++
#if 1
#include<bits/stdc++.h>
#define int long long
using namespace std;
typedef pair<int, int> pp;
const int inf = 2147483647;
signed main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr); cout.tie(nullptr);
    int n, m, s;
    cin >> n >> m >> s;
    vector<vector<pp>> city(n + 1);
    for (int i = 0; i < m; i++)
    {
        int u, v, w;
        cin >> u >> v >> w;
        city[u].push_back({ w,v });
    }
    priority_queue<pp,vector<pp>,greater<pp>> p;
    vector<int> len(n + 1, 2147483647);
    p.push({ 0,s });
    while (!p.empty())
    {
        auto temp = p.top();
        p.pop();
        if (len[temp.second] <= temp.first) { continue; }
        else { len[temp.second] = temp.first; }
        for (auto i : city[temp.second])
        {
            i.first += temp.first;
            if (i.first <= len[i.second])
            {
                p.push(i);
            }
        }
    }
    for (int i = 1; i <= n; i++)
    {
        cout << len[i] << " ";
    }
    system("pause");
    return 0;
}
#endif
```
